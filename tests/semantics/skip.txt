# Corpus lines that run_semantics.py does not compare, as `<regex>  # <reason>`.
# The regex is searched (re.search) in the instruction text, lowercase, as it is
# written in the corpus. A reason is required. Keep this list explicit: it is for
# instructions that depend on state the test cannot give both engines (segment
# descriptors, privilege levels, I/O, the host), not for failures.

# --- segment registers: Unicorn runs with flat, null-selector segments; a load needs a descriptor table
^(lock |rep |repe |repne )?mov (es|cs|ss|ds|fs|gs),  # loads a segment register: needs a GDT/IVT entry, Unicorn and the IR don't model descriptors
^(lock |rep |repe |repne )?pop (es|ss|ds|fs|gs)\b  # loads a segment register
\b(lds|les|lss|lfs|lgs)\b  # loads a segment register and a pointer

# --- far control transfers (load cs)
\b(jmp|call) (far\b|(0x)?[0-9a-f]+:)  # far branch: loads cs from a descriptor
\b(jmp|call) (dword|word|tword) far\b  # far branch through memory
\b(jmp|call) far\b  # far branch
\b(retf|retfw|retfd|retfq|lret|iret|iretw|iretd|iretq)\b  # far return / interrupt return: loads cs, eflags and ss from the stack in a mode-dependent way
\bret far\b  # far return
\b(sysenter|sysexit|syscall|sysret)\b  # needs MSRs

# --- privileged / system instructions
\b(lgdt|lidt|lldt|ltr|sgdt|sidt|sldt|str|lmsw|smsw|clts|invd|wbinvd|invlpg|rsm)\b  # descriptor tables, control registers and caches: not modelled
\b(rdmsr|wrmsr|rdpmc|rdtsc|rdtscp|cpuid|xgetbv|xsetbv|rdrand|rdseed|rdpid)\b  # the result is the host's or the engine's own, not defined by the instruction
\bmov (cr|dr|tr)[0-9]|, (cr|dr|tr)[0-9]  # control, debug and test registers
\b(lar|lsl|verr|verw|arpl)\b  # depend on the descriptor tables
\b(in|out|ins|insb|insw|insd|outs|outsb|outsw|outsd)\b  # port I/O: no device model (Unicorn ignores the access)
\b(hlt|cli|sti|monitor|mwait)\b  # halt / interrupt flag: privileged; monitor/mwait need a monitored address range
^(lock |rep |repe |repne )?(vmcall|vmlaunch|vmresume|vmxoff|vmxon|vmread|vmwrite|vmptrld|vmptrst|vmclear|invept|invvpid|vmrun|vmmcall|vmload|vmsave|stgi|clgi|skinit|invlpga|rdmsrlist|wrmsrlist|swapgs|encls|enclu|enclv|getsec|rdpkru|wrpkru|ud0|ud1|xtest|xbegin|xend|xabort|hreset)\b  # virtualization / SGX / TSX / vendor instructions
